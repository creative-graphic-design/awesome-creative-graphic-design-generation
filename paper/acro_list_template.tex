% Review-oriented research index: show every entry with compact bibliography data.
% Store task/representation annotations in the registry but explain them in tables/prose.
% Never print the verbose task and representation paragraphs in narrow two-column lists.
% Hyperref targets MUST use expandable \acrofield with \ExpandArgs{e} on the id;
% \acrowrite is a print command and produces malformed GoToR/URI PDF annotations.
% Do NOT use description labels for variable-length system names: their label box
% and the fixed body indentation cause mismatched wrap and metadata alignment.
\DeclareAcroProperty{source}
\DeclareAcroProperty{refyear}
\DeclareAcroProperty{venue}
\DeclareAcroProperty{paperurl}
\DeclareAcroProperty{doi}
\DeclareAcroProperty{task}
\DeclareAcroProperty{representation}

\NewAcroTemplate[list]{survey-rich}{%
  \acroheading
  \acropreamble
  \begin{list}{}{%
    \setlength{\leftmargin}{0pt}%
    \setlength{\rightmargin}{0pt}%
    \setlength{\labelwidth}{0pt}%
    \setlength{\labelsep}{0pt}%
    \setlength{\itemindent}{0pt}%
    \setlength{\topsep}{0pt}%
    \setlength{\partopsep}{0pt}%
    \setlength{\parsep}{0pt}%
    \setlength{\itemsep}{.35\baselineskip}%
  }%
    \raggedright
    \hyphenpenalty=10000
    \exhyphenpenalty=10000
    \acronymsmapF{%
      \item\mbox{\textbf{\acrowrite{short}}}\enspace%
      \acrowrite{list}%
      \acroifT{extra}{ (\acrowrite{extra})}%
      \par\nobreak
      {\footnotesize
        \hspace*{1em}%
        \acroifT{refyear}{\textbf{\acrowrite{refyear}}\enspace}%
        \acroifT{venue}{\textit{\acrowrite{venue}}\enspace}%
        \acroifT{source}{Source: \acrowrite{source}\enspace}%
        \acroifT{paperurl}{\href{\ExpandArgs{e}\acrofield{\AcronymID}{paperurl}}{paper}\enspace}%
        \acroifT{doi}{\href{https://doi.org/\ExpandArgs{e}\acrofield{\AcronymID}{doi}}{DOI}\enspace}%
        \acropages
          {\acrotranslate{page}\nobreakspace}
          {\acrotranslate{pages}\nobreakspace}%
      }%
    }{\item\AcroRerun}%
  \end{list}%
}
